cmake_minimum_required(VERSION 3.25)
project(FileBrowserConsumer LANGUAGES CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(PLUGIN_ONLY "Load the QML plugin without linking Quick" OFF)
option(QUICK "Exercise the installed QML module" OFF)
if(QUICK)
  find_package(HolonightFileBrowser REQUIRED COMPONENTS Core Quick)
else()
  find_package(HolonightFileBrowser REQUIRED COMPONENTS Core)
endif()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE HolonightFileBrowser::Core)
if(QUICK)
  target_compile_definitions(consumer PRIVATE QUICK_CONSUMER)
  if(PLUGIN_ONLY)
    target_compile_definitions(consumer PRIVATE PLUGIN_ONLY)
    target_link_libraries(consumer PRIVATE Qt6::Quick)
  else()
    target_link_libraries(consumer PRIVATE HolonightFileBrowser::Quick)
  endif()
endif()
