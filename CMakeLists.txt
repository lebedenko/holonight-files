cmake_minimum_required(VERSION 3.25)
project(holonight-files VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(GNUInstallDirs)
# Presets/tasks supply development defaults; explicit caller paths are authoritative.
set(FILES_RUNTIME_QML_PATH "${QML_IMPORT_PATH}")
string(REPLACE ";" ":" FILES_RUNTIME_QML_PATH "${FILES_RUNTIME_QML_PATH}")
option(BUILD_FILES_APP "Build hn-files and application regressions" ON)
option(BUILD_FILE_BROWSER_QUICK "Build the passive QML listing" ON)
option(BUILD_TESTING "Build tests" OFF)
find_package(Qt6 6.11 REQUIRED COMPONENTS Core)
qt_standard_project_setup(REQUIRES 6.11)
add_subdirectory(libs/file-browser)
if(BUILD_FILES_APP)
  if(NOT BUILD_FILE_BROWSER_QUICK)
    message(FATAL_ERROR "hn-files requires BUILD_FILE_BROWSER_QUICK")
  endif()
  include("${CMAKE_CURRENT_SOURCE_DIR}/tooling/dependencies.cmake")
  find_package(Qt6 6.11 REQUIRED COMPONENTS Gui Quick Qml QuickControls2 Svg)
  find_package(HolonightQt CONFIG REQUIRED)
  find_package(tomlplusplus 3.4 CONFIG REQUIRED)
  find_package(PkgConfig REQUIRED)
  find_package(HolonightImages CONFIG REQUIRED)
  find_package(HolonightThumbnails CONFIG REQUIRED)
  find_package(HolonightSearch CONFIG REQUIRED)
  find_package(HoloNightSystemServices REQUIRED COMPONENTS Storage)
  add_subdirectory(apps/files)
endif()
if(BUILD_TESTING)
  enable_testing()
  find_package(Qt6 6.11 REQUIRED COMPONENTS Test)
  find_package(GTest CONFIG REQUIRED)
  add_subdirectory(tests/file-browser)
  if(BUILD_FILES_APP)
    add_subdirectory(tests)
  endif()
endif()
install(FILES LICENSE LICENSES/GPL-3.0-or-later.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/${PROJECT_NAME})
if(BUILD_FILES_APP)
  install(FILES packaging/org.holonight.Files.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
  install(FILES packaging/org.holonight.Files.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
endif()
find_program(RUN_CLANG_TIDY run-clang-tidy)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
file(GLOB_RECURSE FILES_CXX CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/libs/*.cpp" "${CMAKE_SOURCE_DIR}/apps/*.cpp" "${CMAKE_SOURCE_DIR}/tests/*.cpp")
add_custom_target(format COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/scripts/format-sources.py" VERBATIM)
add_custom_target(format-check COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/scripts/format-sources.py" --check VERBATIM)
add_custom_target(qml-import-check COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/scripts/check-qml-import-policy.py" VERBATIM)
if(BUILD_FILES_APP)
  add_custom_target(qmltypes-check COMMAND bash "${CMAKE_SOURCE_DIR}/scripts/check-qmltypes.sh" "${CMAKE_BINARY_DIR}" DEPENDS hn-files VERBATIM)
  add_custom_target(qml-lint DEPENDS hn-files_qmllint holonight_file_browser_quick_qmllint)
elseif(BUILD_FILE_BROWSER_QUICK)
  add_custom_target(qml-lint DEPENDS holonight_file_browser_quick_qmllint)
endif()
if(RUN_CLANG_TIDY)
  add_custom_target(tidy COMMAND ${RUN_CLANG_TIDY} -p ${CMAKE_BINARY_DIR} -removed-arg=-mno-direct-extern-access -config-file=${CMAKE_SOURCE_DIR}/.clang-tidy ${FILES_CXX})
endif()
